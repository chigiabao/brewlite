\newcommand{\drawactor}[3]{
    \node (#1) at (#2) [minimum width=1.2cm, minimum height=1.8cm] {};
    \draw[thick] (#1.center) ++(0,0.45) circle (0.22cm); % Đầu
    \draw[thick] (#1.center) ++(0,0.23) -- ++(0,-0.55);  % Thân

    \draw[thick] (#1.center) ++(0,-0.32) -- ++(-0.3,-0.45); % Chân trái
    \draw[thick] (#1.center) ++(0,-0.32) -- ++(0.3,-0.45);  % Chân phải
    \node[below=2pt of #1, align=center, font=\bfseries\small] (#1_lbl) {#3};
}
